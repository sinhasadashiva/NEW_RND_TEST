// by counting the number of nodes
#include <bits/stdc++.h>
using namespace std;

class Node {
public:
    int data;
    Node* next;
  
    Node(int x) {
        this->data = x;
        this->next = nullptr;
    }
};

// Helper function to find length of linked list
int getLength(Node* head) {

    // Variable to store length
    int length = 0;

    // Traverse the entire linked list and increment length by
    // 1 for each node
    while (head) {
        length++;
        head = head->next;
    }

    // Return the number of nodes in the linked list
    return length;
}

// Function to find the middle element of the linked list
int reverse_list(Node* base_head,int length) {
    
	Node *head = base_head;
        
    int temp = 1;
    //cout<<"Length::> "<<length<<"\t";
    while(temp<length)  // 1  2  3  4  5
    {
        head = head->next;  
        temp++;
    }
    
        
    //cout<<"Value::> "<<head->data <<endl;
    cout<<head->data <<"\t";
    length = length - 1;
    
    if(length>0)
    {
        reverse_list(base_head, length);
    }
    return 0;
}

void display_nodes(Node* base_head)
{
    Node *temp_node = base_head;
    //printf("Nodes are::>> ");
    while(temp_node!= NULL)
    {
        printf("\t%d",temp_node->data);
        temp_node=temp_node->next;
        //printf("\n");
    }
    printf("\n");
}
int main() {

    // Create a hard-coded linked list:
    // 10 -> 20 -> 30 -> 40 -> 50 -> 60 
    Node* head = new Node(10);
    head->next = new Node(20);
    head->next->next = new Node(30);
    head->next->next->next = new Node(40);
    head->next->next->next->next = new Node(50);
    head->next->next->next->next->next = new Node(60);
    int length = getLength(head);
    cout<<"Display::>> ";
    display_nodes(head);
    cout<<"\nReverse::>> "<<"\t";
    
    reverse_list(head,length);

    return 0;
}
