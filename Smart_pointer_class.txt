 #include<iostream>
using namespace std;

template <class T>

class smart_cls
{
   T *m_ptr ;
   
   public:
   explicit smart_cls(T *tmp_ptr=NULL)
   {
       m_ptr=tmp_ptr;
   }
   ~smart_cls()
   {
       delete m_ptr;
   }
   
   T &operator*()
   {
       return *m_ptr;
   }
   
   T *operator->()
   {
       return m_ptr;
   }
};

int main()
{
    smart_cls<int>obj_ptr(new int()) ;
    *obj_ptr = 100;
    cout<<"Value::>> "<<*obj_ptr<<endl;
    
    smart_cls<string>obj_ptr_1(new string()) ;
    *obj_ptr_1 = "FIS";
    cout<<"String Value::>> "<<*obj_ptr_1;
    return 0;
}
