Input: {1, 0, 3, 0, 5, 0, 6}
Output: {1, 3, 5, 6, 0, 0, 0}


#include <stdio.h>

void swap(int *i,int *j)
{
    *i = *i+*j;
    *j = *i-*j;
    *i = *i-*j;
}

int main()
{
int arr[] = {1, 0, 3, 0, 5, 0, 6};


int len = sizeof(arr)/sizeof(arr[0]);


for(int i = 0; i<len;i++ )
{
	for(int j=i+1; j<len;j++)
	{
		if(arr[i]==0 && arr[j]!=0)
		{
			swap(&arr[i],&arr[j]);
		}
	}
}

for (int i=0;i<len;i++)
{
    printf("%d\t",arr[i]);
}
}
