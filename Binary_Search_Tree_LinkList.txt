#include <iostream>
using namespace std;

class BST
{
public:

    typedef struct Node_Head
    {
        int data;
        struct Node_Head* left;
        struct Node_Head* right;

        Node_Head(int value)
        {
            data = value;
            left = nullptr;
            right = nullptr;
        }
    } Node;

    Node* insert(Node* node, int value)
    {
        if (node == nullptr)
        {
            return new Node_Head(value);
        }

        if (value < node->data)
        {
            node->left = insert(node->left, value);
        }
        else if (value > node->data)
        {
            node->right = insert(node->right, value);
        }

        return node;
    }

    bool search(Node* node, int value)
    {
        if (node == nullptr)
            return false;

        if (node->data == value)
            return true;

        if (value < node->data)
            return search(node->left, value);

        return search(node->right, value);
    }

    void inorder(Node* node)
    {
        if (node == nullptr)
            return;

        inorder(node->left);

        cout << node->data << " ";

        inorder(node->right);
    }
};

int main()
{
    BST tree;

    BST::Node* root_head = nullptr;

    root_head = tree.insert(root_head, 50);
    root_head = tree.insert(root_head, 30);
    root_head = tree.insert(root_head, 70);
    root_head = tree.insert(root_head, 20);
    root_head = tree.insert(root_head, 40);
    root_head = tree.insert(root_head, 60);
    root_head = tree.insert(root_head, 80);

    cout << "In-order traversal: ";
    tree.inorder(root_head);
    cout << endl;

    int value = 40;

    if (tree.search(root_head, value))
        cout << value << " found in BST" << endl;
    else
        cout << value << " not found in BST" << endl;

    value = 100;

    if (tree.search(root_head, value))
        cout << value << " found in BST" << endl;
    else
        cout << value << " not found in BST" << endl;

    return 0;
}
