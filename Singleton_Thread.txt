#include <iostream>
#include <vector>
#include <mutex>
#include <thread>

using namespace std;

class Singleton
{
private:

Singleton()
{
cout<<"Cons"<<endl;
}

~Singleton()
{
cout<<"Desc"<<endl;
}
public:
static Singleton *single;
static Singleton* getInstance();
static Singleton* destroy();


static mutex mute_init;
};
Singleton* Singleton::single = NULL;

mutex Singleton::mute_init;

Singleton* Singleton::getInstance()
{

lock_guard<mutex> lock(mute_init);

if(single == NULL)
{
single = new Singleton();
return single;
}
else
{
return single;
}
}

Singleton* Singleton::destroy()
{
lock_guard<mutex> lock(mute_init);
if(single != NULL)
{
    delete single;
    single = NULL;
}
 return single;
}


int main()
{
vector<thread>threads;

for(auto i =0;i<10;i++)
{
    threads.emplace_back(thread([](){ 
      Singleton::getInstance(); 
      
      Singleton::destroy();
    
    }));
}

for(auto& t: threads)
{
    if(t.joinable())
    {
        t.join();
    }
}

return 0;
}
