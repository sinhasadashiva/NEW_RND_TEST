#include <bits/stdc++.h>
#include <iostream>
#include <vector>
using namespace std;
 
 
void reverseWords(string s)
{
    vector<string> tmp;
    string str = "";
    string final = "";
    for (int i = s.length(); i >=0; i--)
    { 
        if (s[i] == ' ')
        {
            tmp.push_back(str);
            str = "";
        }
        else
            str += s[i];
    }
   
    tmp.push_back(str);

    int left = 0;
    int right = tmp.size() - 1;

    while (left <= right)
    {
       final += tmp[left];

       if (left != right)
       {
          final += " ";
          final += tmp[right];
       }

          final += " ";

          left++;
          right--;
    }

   cout << final <<endl; 
}
 
int main()
{
    string s = "i like this";  //siht ekil i
    //string s = "Hello World I am here now";  //won olleH ereh dlroW ma I
    
    reverseWords(s);
    return 0;
}
