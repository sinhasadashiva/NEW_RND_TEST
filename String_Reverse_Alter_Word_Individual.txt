#include <bits/stdc++.h>
#include <iostream>
#include <vector>
using namespace std;
 
 
void reverseWords(string s)
{
    vector<string> tmp;
    string str = "";
    string final = "";
    for (int i = s.length(); i >=0; i--)
    { 
        if (s[i] == ' ')
        {
            tmp.push_back(str);
            str = "";
        }
        else
            str += s[i];
    }
   
    tmp.push_back(str);

    int no_of_loop = tmp.size()/2;
    int i,j,count=-1;
    
    if(tmp.size() == 3 )
    {
        for(i = 0,j=no_of_loop+2; i<=no_of_loop;i++,j--)
        {
            final += tmp[i];
            final += " ";
            final += tmp[j];
        }
    }
    else if(tmp.size() % 2 != 0)
    {
        for(i = 0,j=no_of_loop+2; i<=no_of_loop;i++,j--)
        {
            if(count == no_of_loop-1)
            {
                final += tmp[i];
                final += " ";
            }
            else
            {
                final += tmp[i];
                final += " ";
                final += tmp[j];
                final += " ";
            }
            count++;
        }
    }
    else
    {
        for(i = 0,j=no_of_loop+2; i<no_of_loop;i++,j--)
        {
                final += tmp[i];
                final += " ";
                final += tmp[j];
                final += " ";
        }
    }
    
//ereh olleH ma dlroW I  
//siht ekil i
//won olleH ereh dlroW ma I
   cout << final <<endl; 
}
 
int main()
{

    string s = "Hello World I am here";
    reverseWords(s);
    return 0;
}
