#include <stdio.h>
#include <malloc.h>
#define ROW     3
#define COL     10

int main(void) 
{
  int *num [ROW][COL];
  int k;
    for(int i = 0 ;i<ROW; i++)
    {
      for(int j = 0 ;j<COL; j++)
      {
        printf("Enter the value::>> "); 
        scanf("%d",&k);
        num[i][j] = (int*) malloc((ROW*COL) * sizeof(int));
        *num[i][j]=k;
      }
    }
  
  for (int i = 0; i < ROW; i++) {
    for (int j = 0; j < COL; j++) 
    {
      printf("%d ", *num[i][j]);
    }
    printf("\n");
  }
  return 0;
}
