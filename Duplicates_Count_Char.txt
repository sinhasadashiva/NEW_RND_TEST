#include <bits/stdc++.h>
#include<iostream>
#include<map>
using namespace std;


int main()
{
    string s1 = "qwerty";
    
    string s2 = "abcdefqqrtgnmjhkrriiiqqrettyellow";
    
    map<char,int> mp;
    
    int len = s2.length();
    
    for(int i=0;i<len;i++)
    {
        mp[s2[i]]++;
    }
    
	for (char ch : s1)
    {
        cout << "Char :: " << ch << " : " << mp[ch] << endl;
    }
return 0;
}
