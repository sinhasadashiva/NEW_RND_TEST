#include <iostream>
#include <algorithm>
using namespace std;

int main()
{
    string s = "abaxyzzyxf";
    string longest = "";

    for (int i = 0; i < s.size(); i++)
    {
        for (int j = i; j < s.size(); j++)
        {
            string x = s.substr(i, j - i + 1);
            string r = x;

            reverse(r.begin(), r.end());

            if (x == r && x.size() > longest.size())
                longest = x;
        }
    }

    cout << longest;

    return 0;
}
