#include <iostream>
#include <string>
#include <cstring>

using namespace std;

string str_rev_func(string str, int len)
{
    string temp;
    
    for(int i = len;i>=0;i--)
    {
        temp += str[i];
    }
    
    return temp;
}

bool str_cmp(string s1, string s2, int len)
{
    bool status = 0;
    
    for(int i = 0;i<len;i++)
    {
        if(s1[i]==s2[i])
        {
            continue;
        }
        else
        {
            status = 1;
            break;
        }
        
    }
    return status;
}

int main()
{
    string s = "12321";
    
    int len = s.length()-1;
    
    cout<<"b4::>> "<<s<<endl;
    
    string str_rev = str_rev_func(s,len);
    
    cout<<"Af::>> "<<str_rev<<endl;
    
    bool status = str_cmp(s,str_rev,len);
    
    if(status == 0)
    {
        cout<<"Palindrome"<<endl;
    }
    else
    {
        cout<<"Not Palindrome"<<endl;
    }


    return 0;
}
