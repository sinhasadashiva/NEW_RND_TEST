#include<stdio.h>
#include<malloc.h>

typedef struct school
{
int roll_no;
struct school *nxt_ptr;
} student;

student *HEAD = NULL;

void create(int x)
{
	student *new_node = (student*)malloc(sizeof(student));
	new_node->roll_no = x;
	new_node->nxt_ptr = HEAD;
	HEAD = new_node;
}

void del_beg()
{ 
	student *temp_node = HEAD;
	HEAD = temp_node->nxt_ptr;
	free(temp_node);
} 

void add_midd(int x, int pos)
{
	student *new_node = (student*)malloc(sizeof(student));
	student *temp_node = HEAD;
	new_node->roll_no = x;
	int temp_pos = 1;
	while(temp_node->nxt_ptr !=NULL && temp_pos < pos)
	{
		temp_node = temp_node->nxt_ptr;
		temp_pos++;
	}
	
	new_node->nxt_ptr = temp_node->nxt_ptr;
	temp_node->nxt_ptr = new_node;
}

void del_midd(int pos)
{
	student *midd_node = HEAD;
	int temp_pos = 1;
	
	student *temp_node;
	
	while(midd_node->nxt_ptr !=NULL && temp_pos < pos-1)
	{
	    midd_node = midd_node->nxt_ptr;
		temp_pos++;
	}
	
	if(pos == 1)
	{
	    temp_node = HEAD;
	    HEAD = temp_node->nxt_ptr;
	    free(temp_node);
	}
	else
	{
		//   2    
		// 1   3 4 5
		
	    temp_node = midd_node->nxt_ptr;
	    midd_node->nxt_ptr = temp_node->nxt_ptr;
	    free(temp_node);
	}
}


void add_last(int x)
{
	student *new_node = (student*)malloc(sizeof(student));
	
	student *temp_node = HEAD;

	while(temp_node->nxt_ptr !=NULL)
	{
	    temp_node = temp_node->nxt_ptr;
	}
	
	new_node->roll_no = x;
	temp_node->nxt_ptr = new_node;
	new_node->nxt_ptr = NULL;
}


void del_last()
{
	student *temp_node = HEAD;
	
	student *last_node;
	
	while(temp_node->nxt_ptr->nxt_ptr!=NULL)
	{
		temp_node = temp_node->nxt_ptr;
	}
	
	last_node = temp_node->nxt_ptr;
	temp_node->nxt_ptr = NULL;
	free(last_node);
}

void display_nodes()
{
    student *temp_node = HEAD;
    printf("Nodes are::>> ");
    while(temp_node!= NULL)
    {
        printf("\t%d",temp_node->roll_no);
        temp_node=temp_node->nxt_ptr;
        //printf("\n");
    }
    printf("\n");
}
void delete_small(int small_val)
{
   student* Temp_obj = HEAD;// = new List();
   student *temp_node;
   student *temp_node_prev=NULL;
  
  
  while(Temp_obj!= NULL)
  {
      if(Temp_obj->roll_no == small_val)
      {
        if(Temp_obj == HEAD)
        {
            HEAD = Temp_obj->nxt_ptr;
            free(Temp_obj);
            break;
        }
        
        temp_node = Temp_obj;  //  3 6 7 17
        Temp_obj = temp_node_prev;
        Temp_obj->nxt_ptr = temp_node->nxt_ptr;
        free(temp_node);
      }
      else
      {
          printf("else\n");
          temp_node_prev = Temp_obj;
          Temp_obj = Temp_obj->nxt_ptr;
      }
  }
}

void check()
{
  int largest, smallest;
  
  student* Temp_obj = HEAD;// = new List();
  
  while(Temp_obj != NULL)
  {
     if(largest<Temp_obj->roll_no)
     {
     	largest = Temp_obj->roll_no;
     }
     
    
     if(smallest>Temp_obj->roll_no)
     {
     	smallest = Temp_obj->roll_no;
     }
     
     Temp_obj = Temp_obj->nxt_ptr;
   }
  printf("Largest::>> %d & Smallest::>> %d\n\n",largest,smallest);
  delete_small(smallest);
    
}

void seq_insert(int x)
{
    student *new_node = (student*)malloc(sizeof(student));
	
	student *temp_node = HEAD;
	
	new_node->roll_no = x;

	while(temp_node !=NULL)
	{
	    if(temp_node->nxt_ptr == NULL)
	    {
	      temp_node->nxt_ptr = new_node;
	      new_node->nxt_ptr = NULL;
	    }
	    temp_node = temp_node->nxt_ptr;
	}
}

void insert(int x)
{
    student* Temp_Val = HEAD;
    
    student* Temp_Val_next = (student*)malloc(sizeof(student));
    
    while(Temp_Val != NULL)  // (5) 6 7 9 12 17
    {
        if(Temp_Val->roll_no>x && Temp_Val == HEAD)
        {
            student* Temp_Val_Curr_head = (student*)malloc(sizeof(student));
            Temp_Val_Curr_head->roll_no = x;
            Temp_Val_Curr_head->nxt_ptr = HEAD;
            HEAD = Temp_Val_Curr_head;
            //create(x);
        }
        else
        {
            if(Temp_Val->nxt_ptr != NULL)
               Temp_Val_next = Temp_Val->nxt_ptr;
            
            if(Temp_Val->roll_no<x && Temp_Val_next->roll_no >x) //  (5) 6 7 9 12 (15) 17 (20)
            {
                 student* Temp_Val_Curr = (student*)malloc(sizeof(student));   
                 
                 Temp_Val_Curr->roll_no = x;
                 
                 Temp_Val_Curr->nxt_ptr = Temp_Val->nxt_ptr;
                 
                 Temp_Val->nxt_ptr = Temp_Val_Curr;
                 
            } 
           
            else if(Temp_Val->roll_no<x && Temp_Val->nxt_ptr == NULL)
            {
                student* Temp_Val_Curr = (student*)malloc(sizeof(student)); 
                
                Temp_Val_Curr->roll_no = x;
                
                Temp_Val->nxt_ptr = Temp_Val_Curr;
                
                Temp_Val_Curr->nxt_ptr = NULL;
            }
        }
        
        Temp_Val = Temp_Val->nxt_ptr;
        
    }
}
