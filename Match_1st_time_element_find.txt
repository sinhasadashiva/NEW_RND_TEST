#include <iostream>

using namespace std;

bool find_job(int A[],int start_point)
{
int total=0, total_sum=12;
bool return_val = false;
   
   for(int i =start_point;i<5;i++)
   {
       if(total<total_sum)
       {
            total = total+A[i] ;
           
            if(total > total_sum)
            {
                return_val = true;
            }
            if(total == total_sum)
            {
                cout << "Total::>> "<<total<<endl;
                cout << "1st Index_no::>> "<<start_point+1<<endl;
                cout<<"2nd Index_no::>> "<<i+1<<endl;
                return_val = false;
            }
       }
   
   }
   
return return_val;
}

int main() {
   
   // N = 5, S = 12
   int A[] = {1,3,3,7,5};
   
   int start_p = 0;
   bool j = true;
   
   while(j)
   {
       j = find_job(A,start_p);
       start_p++;
   }
return 0;
}
