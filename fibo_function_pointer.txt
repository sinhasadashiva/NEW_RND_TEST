#include <iostream>

using namespace std;

int fibo(int start, int second, int num)
{
    if(num>0)
    {
        cout<<start<<" ";
        fibo(second, start+second, num-1);
    }
    
    return 0;
}

int main()
{
    //std::cout<<"Hello World";
    
    int start = 0, second = 1, n;
    
    cout<<"Enter Number:: "<<endl;
    
    cin>>n;

    
    
    int (*x)(int,int,int);
    
    x = &fibo;

    x(start,second,n);

    return 0;
}
