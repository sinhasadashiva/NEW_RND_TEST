   class link_list
   {
     typedef struct Node_list
     {
        int value;
        Node_list *nxt;
     } Node_Obj;
     
     mutex mutex_obj;
     
     Node_Obj *HEAD;
     
     public:
     
     link_list()
     {
         HEAD = NULL;
     }
     
     void create(int x)
     {
       lock_guard<mutex>mutex_locker(mutex_obj);
	   
       Node_Obj *temp = new Node_Obj();
       
       temp->value = x;
       
       temp->nxt = HEAD;
       
       HEAD = temp;
     }
     
     void display()
     {
         lock_guard<mutex>mutex_locker(mutex_obj);
		 
         Node_Obj *temp = HEAD;
         
         while(temp != NULL)
         {
             cout<<temp->value<<"  ";
             temp=temp->nxt;
         }
     }
   };
