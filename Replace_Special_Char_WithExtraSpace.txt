Write a C program to replace special characters in a string with two spaces.
Do not use any temp variables. The string is having enough space to accommodate the double spaces.
 
Input: This@is#an$example%program
Output: This  is  an  example  program


#include <stdio.h>
#include <string.h>

int main()
{
    char str[100] = "This@is#an$example%program";

    int len = sizeof(str)/sizeof(str[0]);
    
    for (int i = 0; str[i] != '\0'; i++)
    {
        if (str[i] == '@' ||
            str[i] == '#' ||
            str[i] == '$' ||
            str[i] == '%')
        {
            // Shift characters to the right
            for (int j = strlen(str); j >= i; j--)
            {
                str[j + 1] = str[j];
            }
			
			//memmove(&str[i + 2], &str[i + 1], len - i);

            str[i] = ' ';
            str[i + 1] = ' ';

            i++;   // skip the second space
        }
    }

    printf("%s\n", str);

    return 0;
}
