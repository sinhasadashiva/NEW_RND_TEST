// Cpp to find first repeating element
// using Naive approach in O(n^2) Time and O(1) Space
#include <bits/stdc++.h>
using namespace std;

int firstRepeatingElement(int arr[],int n)
{
    
    // Nested loop to check for repeating elements
    for (int i = 0; i < n; i++) 
	{
        for (int j = i + 1; j < n; j++) 
		{
            if (arr[i] == arr[j]) 
			{
                cout << "First repeating value is " << arr[i] << endl;
                return 0;
            }
        }
    }

    // If no repeating element is found, return -1
    cout << "No repeating found!" << endl;
    return 0;
}

int main()
{
    int arr[] = {10, 5, 3, 4, 3, 5, 6};
    int n = sizeof(arr) / sizeof(arr[0]);
    cout<<"value::>>> "<<n<<endl;
    int index = firstRepeatingElement(arr,n);
    
    return 0;
}
