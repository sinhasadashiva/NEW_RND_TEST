:g/^$/d (in the file) delete empty lines in the file

sed -i s/old string/new string/g filename

find . -type f | xargs grep -i "string_name" (find the string in a folder files and sub folder files also)

ls | head -1 (first line means 1st file)

ls | tail -1 (last line means last file)

find . -iname filename (case not sensitive)

find . -name filename (case sensitive)

cat | rev filename (all lines are reverse format)

tac filename (descending format)

tail -F filename (we can see the changes without opened file)

set nu (display line numbers)

:1,3co2 (copy from 1st line 3rd line and paste after 2nd line)

ls -d */ display only folders
