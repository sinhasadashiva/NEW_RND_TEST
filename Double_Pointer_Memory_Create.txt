#include <stdio.h>
#include <stdlib.h>

int main()
{
    int (*ptr)[3];

    ptr = malloc(2 * sizeof(int));

    if (ptr == NULL)
        return 1;

    ptr[0][0] = 10;
    ptr[0][1] = 20;
    ptr[0][2] = 30;

    ptr[1][0] = 40;
    ptr[1][1] = 50;
    ptr[1][2] = 60;

    for (int i = 0; i < 2; i++)
    {
        for (int j = 0; j < 3; j++)
        {
            printf("%d ", ptr[i][j]);
        }
        printf("\n");
    }

    free(ptr);

    return 0;
}
